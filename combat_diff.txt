+def _apply_casualties(troop_rows: list[dict], lost_power: float, power_field: str) -> None:
+    """Mutates troop_rows in place, setting a 'survivors' key on each, distributing

+    if total_power <= 0 or lost_power <= 0:
+        for r in troop_rows:

+        return
